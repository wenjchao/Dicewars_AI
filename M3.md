```python
"""Core game state management."""

import time
from dataclasses import dataclass, field
from typing import List, Dict, Optional, Any
from ..api.types import GamePhase, GameState, GameConfig, MapLayout
from .hash import state_hash, position_hash, layout_hash


@dataclass
class CoreState:
    """Internal core game state."""
    
    # Board state
    owners: List[int]  # Territory owners (player_id or -1 for neutral)
    dice: List[int]  # Dice per territory
    
    # Game metadata  
    active_player_id: int
    phase: GamePhase
    turn_number: int
    tick_id: int
    eliminated_players: List[int] = field(default_factory=list)
    winner_id: Optional[int] = None
    
    # RNG state
    rng_counters: Dict[str, int] = field(default_factory=lambda: {
        "setup": 0,
        "turn": 0, 
        "battle": 0
    })
    
    # Static configuration (for hashing)
    _config: GameConfig = field(repr=False)
    _layout_hash: int = field(init=False, repr=False)
    
    def __post_init__(self):
        """Initialize derived fields."""
        self._layout_hash = layout_hash(self._config.map_layout)
    
    @classmethod
    def create_initial(cls, config: GameConfig) -> 'CoreState':
        """Create initial game state from config."""
        num_territories = len(config.map_layout.territories)
        
        # Initialize with neutral territories
        owners = [-1] * num_territories
        dice = [0] * num_territories
        
        return cls(
            owners=owners,
            dice=dice,
            active_player_id=0,  # Will be set by setup
            phase=GamePhase.SETUP,
            turn_number=0,
            tick_id=0,
            _config=config
        )
    
    def increment_tick(self) -> None:
        """Increment tick counter."""
        self.tick_id += 1
    
    def compute_state_hash(self) -> int:
        """Compute current state hash."""
        return state_hash(
            owners=self.owners,
            dice=self.dice,
            active_player_id=self.active_player_id,
            phase=self.phase.value,
            turn_number=self.turn_number,
            winner_id=self.winner_id,
            tick_id=self.tick_id,
            rng_counters=self.rng_counters,
            layout_hash=self._layout_hash,
            rules_version=self._config._rules_version,  # From about.py
            schema_version=self._config._schema_version,
            battle_system=self._config.battle_system.value,
            supply_policy=self._config.supply_policy.value,
            max_dice_per_territory=self._config.max_dice_per_territory
        )
    
    def compute_position_hash(self) -> int:
        """Compute current position hash (excludes tick_id, RNG)."""
        return position_hash(
            owners=self.owners,
            dice=self.dice,
            active_player_id=self.active_player_id,
            phase=self.phase.value,
            turn_number=self.turn_number,
            winner_id=self.winner_id,
            layout_hash=self._layout_hash,
            rules_version=self._config._rules_version,
            schema_version=self._config._schema_version,
            battle_system=self._config.battle_system.value,
            supply_policy=self._config.supply_policy.value,
            max_dice_per_territory=self._config.max_dice_per_territory
        )
    
    def to_public(self) -> GameState:
        """Convert to public GameState DTO."""
        return GameState(
            owners=self.owners.copy(),
            dice=self.dice.copy(),
            active_player_id=self.active_player_id,
            phase=self.phase,
            turn_number=self.turn_number,
            tick_id=str(self.tick_id),  # v5: decimal string
            winner_id=self.winner_id,
            eliminated_players=self.eliminated_players.copy()
        )
    
    def advance_to_next_player(self) -> bool:
        """
        Advance to next active player.
        Returns True if turn_number was incremented.
        """
        original_turn = self.turn_number
        
        # Find next alive player
        num_players = self._config.num_players
        current = self.active_player_id
        
        for _ in range(num_players):
            current = (current + 1) % num_players
            if current not in self.eliminated_players:
                self.active_player_id = current
                break
        else:
            # All players eliminated - shouldn't happen
            raise RuntimeError("No alive players found")
        
        # Increment turn when we cycle back to a player with lower ID
        # or when we reach the original player
        if self.active_player_id <= (self.active_player_id - 1) % num_players:
            self.turn_number += 1
            return True
        
        return False
```

```python
"""Public GameEngine implementation."""

from typing import Union, List
from ..api.types import (
    GameConfig, Capabilities, TurnContext, ActionResult,
    AttackAction, EndTurnAction, DistributeSupplyAction, PenaltyAction,
    SupplyDescriptor, InvalidAction
)
from ..api.engine import GameEngine as GameEngineInterface
from ..core.state import CoreState
from ..core.attack_pairs import AttackPairs
from ..core.hash import layout_hash
from ..__about__ import RULES_VERSION, SCHEMA_VERSION


class GameEngine(GameEngineInterface):
    """Public GameEngine implementation."""
    
    def __init__(self, config: GameConfig) -> None:
        """Initialize engine with configuration."""
        self.config = config
        self.state = CoreState.create_initial(config)
        self.attack_pairs = AttackPairs.from_map_layout(config.map_layout)
        
        # Start with player 0 in ATTACK phase for walking skeleton
        self.state.phase = GamePhase.ATTACK
        self.state.active_player_id = 0
    
    def get_capabilities(self) -> Capabilities:
        """Get engine capabilities."""
        return Capabilities(
            rules_version=RULES_VERSION,
            schema_version=SCHEMA_VERSION,
            map_name=self.config.map_layout.map_name,
            num_players=self.config.num_players,
            num_territories=len(self.config.map_layout.territories),
            layout_hash=str(layout_hash(self.config.map_layout)),  # v5: decimal string
            action_index_hash=str(self.attack_pairs.hash64),       # v5: decimal string
            winner_policy=self.config.winner_policy.value,
            winner_tiebreak_chain=[t.value for t in self.config.winner_tiebreak_chain],
            max_turns=self.config.max_turns,
            max_dice_per_territory=self.config.max_dice_per_territory,
            supports_delta=False,
            supports_penalties=True,
            uint64_encoding="decimal"
        )
    
    def get_turn_context(self) -> TurnContext:
        """Get current turn context (pure read - no state changes)."""
        if self.state.phase == GamePhase.GAME_OVER:
            return TurnContext(
                game_state=self.state.to_public(),
                valid_actions=[]  # No actions available
            )
        
        # For M3 walking skeleton: only EndTurnAction available
        valid_actions = [EndTurnAction()]
        
        # Supply descriptor not implemented yet (M5)
        supply = None
        
        return TurnContext(
            game_state=self.state.to_public(),
            valid_actions=valid_actions,
            supply=supply,
            time_remaining_ms=None  # Timeout handling in M6
        )
    
    def apply_action(
        self, 
        action: Union[AttackAction, EndTurnAction, DistributeSupplyAction]
    ) -> ActionResult:
        """Apply player action."""
        
        # Game over check
        if self.state.phase == GamePhase.GAME_OVER:
            return self._create_error_result(InvalidAction.BAD_PHASE, "Game is over")
        
        # For M3: only handle EndTurnAction
        if isinstance(action, EndTurnAction):
            return self._apply_end_turn()
        
        # Other actions not implemented yet
        if isinstance(action, AttackAction):
            return self._create_error_result(InvalidAction.BAD_PHASE, "Attacks not implemented yet (M4)")
        
        if isinstance(action, DistributeSupplyAction):
            return self._create_error_result(InvalidAction.BAD_PHASE, "Supply not implemented yet (M5)")
        
        return self._create_error_result(InvalidAction.UNKNOWN_ACTION_TYPE, f"Unknown action type: {type(action)}")
    
    def apply_penalty(self, reason: str, penalty_type: str) -> ActionResult:
        """Apply penalty action (v5)."""
        # M6 implementation - for now just end turn
        self.state.increment_tick()
        
        # Advance player but don't increment turn number (penalty doesn't advance turn)
        self.state.advance_to_next_player()  # Ignore return value for penalties
        
        penalty_action = PenaltyAction(reason=reason, penalty_type=penalty_type)
        
        return ActionResult(
            success=True,
            tick_id=str(self.state.tick_id),
            state_hash=str(self.state.compute_state_hash()),
            position_hash=str(self.state.compute_position_hash()),
            new_game_state=self.state.to_public(),
            battle_result=None,
            error_code=None,
            error_data=None,
            delta=None
        )
    
    def _apply_end_turn(self) -> ActionResult:
        """Apply EndTurnAction."""
        self.state.increment_tick()
        
        # Advance to next player
        turn_incremented = self.state.advance_to_next_player()
        
        # Check max turns
        if self.state.turn_number >= self.config.max_turns:
            self.state.phase = GamePhase.GAME_OVER
            # Winner determination in M7
        
        return ActionResult(
            success=True,
            tick_id=str(self.state.tick_id),
            state_hash=str(self.state.compute_state_hash()),
            position_hash=str(self.state.compute_position_hash()),
            new_game_state=self.state.to_public(),
            battle_result=None,
            error_code=None,
            error_data=None,
            delta=None
        )
    
    def _create_error_result(self, error_code: InvalidAction, message: str) -> ActionResult:
        """Create error ActionResult."""
        return ActionResult(
            success=False,
            tick_id=str(self.state.tick_id),
            state_hash=str(self.state.compute_state_hash()),
            position_hash=str(self.state.compute_position_hash()),  # Always present
            new_game_state=self.state.to_public(),
            battle_result=None,
            error_code=error_code.value,
            error_data={"message": message},
            delta=None
        )
```

```python
"""Game coordinator loop implementation."""

import time
from typing import List, Optional
from ..api.types import GameConfig, GameState, GamePhase
from ..api.client import PlayerClient
from ..api.observer import GameObserver
from ..engine.public import GameEngine


class GameCoordinator:
    """Game coordinator orchestrator."""
    
    def __init__(
        self, 
        config: GameConfig,
        clients: List[PlayerClient],
        observers: Optional[List[GameObserver]] = None
    ) -> None:
        """Initialize coordinator."""
        if len(clients) != config.num_players:
            raise ValueError(f"Expected {config.num_players} clients, got {len(clients)}")
        
        self.config = config
        self.clients = clients
        self.observers = observers or []
        self.engine = GameEngine(config)
        
        # Track timeout windows (M6)
        self._timeout_start: Optional[float] = None
    
    def run(self) -> GameState:
        """Run game to completion."""
        initial_state = self.engine.get_turn_context().game_state
        
        # Notify observers of game start
        for observer in self.observers:
            observer.on_game_start(self.config, initial_state)
        
        # Notify clients of game start
        for i, client in enumerate(self.clients):
            client.on_game_start(self.config, i)
        
        # Main game loop
        while True:
            current_state = self.engine.get_turn_context().game_state
            
            if current_state.phase == GamePhase.GAME_OVER:
                break
            
            # Notify observers of turn start
            # Only when a new alive player enters ATTACK phase
            if current_state.phase == GamePhase.ATTACK:
                for observer in self.observers:
                    observer.on_turn_start(current_state.active_player_id, current_state.turn_number)
            
            # Get action from active player
            try:
                context = self.engine.get_turn_context()
                active_client = self.clients[current_state.active_player_id]
                
                # Start timeout window (M6 - stub for now)
                self._timeout_start = time.time()
                
                action = active_client.get_action(context)
                
                # Reset timeout
                self._timeout_start = None
                
                # Apply action
                result = self.engine.apply_action(action)
                
                if result.success:
                    # Notify observers of successful action
                    for observer in self.observers:
                        observer.on_action_executed(action, result)
                else:
                    # Notify observers of invalid action
                    for observer in self.observers:
                        observer.on_invalid_action(action, result.error_code, result.error_data)
                    
                    # Apply penalty (M6 - simplified for M3)
                    penalty_result = self.engine.apply_penalty("invalid_action", "auto_end_turn")
                    
                    # Notify observers of penalty
                    penalty_action = PenaltyAction(reason="invalid_action", penalty_type="auto_end_turn")
                    for observer in self.observers:
                        observer.on_action_executed(penalty_action, penalty_result)
            
            except Exception as e:
                # Handle client errors (M6 - simplified)
                print(f"Client error: {e}")
                penalty_result = self.engine.apply_penalty("client_error", "auto_end_turn")
                
                penalty_action = PenaltyAction(reason="client_error", penalty_type="auto_end_turn")
                for observer in self.observers:
                    observer.on_action_executed(penalty_action, penalty_result)
        
        # Game over
        final_state = self.engine.get_turn_context().game_state
        
        # Notify observers
        for observer in self.observers:
            observer.on_game_end(final_state)
        
        # Notify clients
        for client in self.clients:
            client.on_game_end(final_state)
        
        return final_state
```

```python
"""Base observer implementations."""

from typing import Optional, Union
from ..api.observer import GameObserver
from ..api.types import (
    GameConfig, GameState, ActionResult,
    AttackAction, EndTurnAction, DistributeSupplyAction, PenaltyAction
)


class LoggingObserver(GameObserver):
    """Simple logging observer for debugging."""
    
    def __init__(self, verbose: bool = False):
        self.verbose = verbose
    
    def on_game_start(self, config: GameConfig, initial_state: GameState) -> None:
        """Called when game starts."""
        print(f"Game started: {config.map_layout.map_name}, {config.num_players} players")
        if self.verbose:
            print(f"Initial state: tick={initial_state.tick_id}, phase={initial_state.phase.value}")
    
    def on_turn_start(self, player_id: int, turn_number: int) -> None:
        """Called when a new alive player enters ATTACK phase."""
        print(f"Turn {turn_number}: Player {player_id} starting ATTACK phase")
    
    def on_action_executed(
        self, 
        action: Union[AttackAction, EndTurnAction, DistributeSupplyAction, PenaltyAction],
        result: ActionResult
    ) -> None:
        """Called after any action is executed."""
        if isinstance(action, PenaltyAction):
            print(f"PENALTY: {action.reason} -> {action.penalty_type} (tick={result.tick_id})")
        else:
            print(f"Action: {action.type} -> {'SUCCESS' if result.success else 'FAILED'} (tick={result.tick_id})")
        
        if self.verbose and result.battle_result:
            br = result.battle_result
            print(f"  Battle: {br.attacker_rolls} vs {br.defender_rolls} -> {br.winner}")
    
    def on_invalid_action(
        self,
        action: Union[AttackAction, EndTurnAction, DistributeSupplyAction],
        error_code: str,
        error_data: Optional[dict] = None
    ) -> None:
        """Called when an invalid action is attempted."""
        print(f"INVALID: {action.type} -> {error_code}")
        if self.verbose and error_data:
            print(f"  Error data: {error_data}")
    
    def on_game_end(self, final_state: GameState) -> None:
        """Called when game ends."""
        print(f"Game ended: tick={final_state.tick_id}, winner={final_state.winner_id}")


class SilentObserver(GameObserver):
    """Silent observer that just tracks events."""
    
    def __init__(self):
        self.events = []
    
    def on_game_start(self, config: GameConfig, initial_state: GameState) -> None:
        self.events.append(("game_start", config.map_layout.map_name, initial_state.tick_id))
    
    def on_turn_start(self, player_id: int, turn_number: int) -> None:
        self.events.append(("turn_start", player_id, turn_number))
    
    def on_action_executed(
        self, 
        action: Union[AttackAction, EndTurnAction, DistributeSupplyAction, PenaltyAction],
        result: ActionResult
    ) -> None:
        self.events.append(("action", action.type, result.success, result.tick_id))
    
    def on_invalid_action(
        self,
        action: Union[AttackAction, EndTurnAction, DistributeSupplyAction],
        error_code: str,
        error_data: Optional[dict] = None
    ) -> None:
        self.events.append(("invalid", action.type, error_code))
    
    def on_game_end(self, final_state: GameState) -> None:
        self.events.append(("game_end", final_state.tick_id, final_state.winner_id))
```
