# M7 — Winner & Game Over

## Overview
Implement complete winner determination system with elimination detection, max-turns boundaries, and deterministic tiebreaker chains. This milestone focuses on v5 winner policies with same-mutation elimination victory and configurable tiebreak systems.

## Goals
- Implement game over detection: elimination victory and max-turns boundary
- Build v5 winner policies: `NONE`, `MOST_TERRITORIES`, `MOST_TOTAL_DICE`, `LARGEST_CONNECTED`, `LEXICOGRAPHIC`
- Handle deterministic tiebreaker chains with configurable ordering
- Ensure elimination victory sets `GAME_OVER` in same mutation (no extra tick)
- Handle max-turns precedence over pending penalties

## Architecture

### Winner Policy System
```python
# v5 Winner Policies:
# - NONE: Allows winner_id = None (draws permitted)  
# - MOST_TERRITORIES: Player with most territories wins
# - MOST_TOTAL_DICE: Player with highest total dice wins
# - LARGEST_CONNECTED: Player with largest connected region wins
# - LEXICOGRAPHIC: Lowest player_id wins (final tiebreaker)

# Tiebreak Chain: [PRIMARY, SECONDARY, ..., LEXICOGRAPHIC]
# Each policy applied in sequence until unique winner found
```

### Game Over Conditions
1. **Elimination Victory**: Only one player remains with territories
2. **Max-turns Boundary**: Turn limit reached, winner determined by policy
3. **Precedence**: Max-turns applied before pending auto-skip mutations

### Winner Determination Flow
```python
# Game over triggers:
# 1. Check elimination (same mutation as conquest)
# 2. Check max-turns boundary (before turn advancement) 
# 3. Apply winner policy chain
# 4. Set phase = GAME_OVER, winner_id determined
# 5. get_turn_context() returns empty valid_actions
```

## Implementation Tasks

### Phase 1: Winner Policy Infrastructure

1. **Create `core/winner.py` with policy implementations**:
   ```python
   from enum import Enum
   from typing import Optional, List, Set
   
   class WinnerPolicy(Enum):
       NONE = "none"                    # v5: allows winner_id = None
       MOST_TERRITORIES = "most_territories"
       MOST_TOTAL_DICE = "most_total_dice" 
       LARGEST_CONNECTED = "largest_connected"
       LEXICOGRAPHIC = "lexicographic"
   
   class WinnerDeterminer:
       def __init__(self, state: CoreState, config: GameConfig):
           self.state = state
           self.config = config
           
       def determine_winner(self) -> Optional[int]:
           """Apply winner policy chain to determine winner."""
           
           active_players = self._get_active_players()
           
           # Single player = automatic winner (elimination)
           if len(active_players) == 1:
               return active_players[0]
               
           # Apply tiebreak chain
           for policy in self.config.winner_tiebreak_chain:
               winner = self._apply_policy(policy, active_players)
               if winner is not None:
                   return winner
                   
           # Final fallback: lexicographic (lowest ID)
           return min(active_players) if active_players else None
   ```

2. **Implement individual winner policies**:
   ```python
   def _apply_policy(self, policy: WinnerPolicy, candidates: List[int]) -> Optional[int]:
       """Apply single winner policy to candidate players."""
       
       if policy == WinnerPolicy.NONE:
           return None  # v5: explicit draw support
           
       elif policy == WinnerPolicy.MOST_TERRITORIES:
           return self._winner_by_territories(candidates)
           
       elif policy == WinnerPolicy.MOST_TOTAL_DICE:
           return self._winner_by_total_dice(candidates)
           
       elif policy == WinnerPolicy.LARGEST_CONNECTED:
           return self._winner_by_connected_regions(candidates)
           
       elif policy == WinnerPolicy.LEXICOGRAPHIC:
           return min(candidates) if candidates else None
           
       return None
   
   def _winner_by_territories(self, candidates: List[int]) -> Optional[int]:
       """Winner by most territories owned."""
       territory_counts = {}
       for player_id in candidates:
           territory_counts[player_id] = sum(
               1 for owner in self.state.owners if owner == player_id
           )
           
       max_count = max(territory_counts.values())
       winners = [p for p, count in territory_counts.items() if count == max_count]
       
       return winners[0] if len(winners) == 1 else None  # Tie = no winner
   
   def _winner_by_total_dice(self, candidates: List[int]) -> Optional[int]:
       """Winner by most total dice owned.""" 
       dice_totals = {}
       for player_id in candidates:
           dice_totals[player_id] = sum(
               dice for i, dice in enumerate(self.state.dice) 
               if self.state.owners[i] == player_id
           )
           
       max_total = max(dice_totals.values())
       winners = [p for p, total in dice_totals.items() if total == max_total]
       
       return winners[0] if len(winners) == 1 else None
   ```

3. **Implement connected regions analysis**:
   ```python
   def _winner_by_connected_regions(self, candidates: List[int]) -> Optional[int]:
       """Winner by largest connected territory region."""
       
       region_sizes = {}
       for player_id in candidates:
           player_territories = [
               i for i, owner in enumerate(self.state.owners) 
               if owner == player_id
           ]
           region_sizes[player_id] = self._largest_connected_region_size(
               player_territories
           )
           
       max_size = max(region_sizes.values())
       winners = [p for p, size in region_sizes.items() if size == max_size]
       
       return winners[0] if len(winners) == 1 else None
   
   def _largest_connected_region_size(self, territories: List[int]) -> int:
       """Find size of largest connected region for territory set."""
       if not territories:
           return 0
           
       # Build adjacency graph for player's territories
       territory_set = set(territories)
       adjacency_graph = {}
       
       for territory_id in territories:
           adjacency_graph[territory_id] = [
               adj for adj in self.config.map_layout.territories[territory_id].adjacencies
               if adj in territory_set
           ]
       
       # Find largest connected component using DFS
       visited = set()
       max_component_size = 0
       
       for territory_id in territories:
           if territory_id not in visited:
               component_size = self._dfs_component_size(
                   territory_id, adjacency_graph, visited
               )
               max_component_size = max(max_component_size, component_size)
               
       return max_component_size
   ```

### Phase 2: Game Over Detection

4. **Update elimination detection in `core/state.py`**:
   ```python
   def check_elimination_victory(self) -> bool:
       """Check for elimination victory (same mutation)."""
       
       active_players = self._get_active_players()
       
       if len(active_players) <= 1:
           # Elimination victory - set game over immediately
           self.phase = GamePhase.GAME_OVER
           
           # Determine winner
           winner_determiner = WinnerDeterminer(self, self.config)
           self.winner_id = winner_determiner.determine_winner()
           
           return True
           
       return False
   
   def _get_active_players(self) -> List[int]:
       """Get players who still own territories."""
       active = set()
       for owner in self.owners:
           if owner >= 0:  # Valid player ID
               active.add(owner)
       return list(active)
   ```

5. **Update max-turns boundary detection**:
   ```python
   def check_max_turns_boundary(self) -> bool:
       """Check max-turns boundary with precedence over penalties."""
       
       if self.turn_number >= self.config.max_turns:
           # v5: Max-turns precedence - cancel pending penalties
           if hasattr(self, 'pending_skips'):
               self.pending_skips.clear()
               
           # Set game over
           self.phase = GamePhase.GAME_OVER
           
           # Determine winner by policy
           winner_determiner = WinnerDeterminer(self, self.config)
           self.winner_id = winner_determiner.determine_winner()
           
           return True
           
       return False
   ```

### Phase 5: Observer Integration

10. **Update coordinator for game over events**:
    ```python
    def _check_game_over(self) -> bool:
        """Check all game over conditions."""
        
        # Check elimination first (during battle mutations) 
        if self.state.phase == GamePhase.GAME_OVER:
            self._notify_game_over()
            return True
            
        # Check max-turns boundary (before turn advancement)
        if self.state.check_max_turns_boundary():
            self._notify_game_over()
            return True
            
        return False
    
    def _notify_game_over(self):
        """Notify observers of game over."""
        for observer in self.observers:
            observer.on_game_end(self.state.to_public_state())
    ```
